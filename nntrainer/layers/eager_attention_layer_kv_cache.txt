EagerAttentionLayer::EagerAttentionLayer() {
  wt_idx.fill(std::numeric_limits<unsigned>::max());
}

EagerAttentionLayer::~EagerAttentionLayer() {}

void EagerAttentionLayer::finalizeCommon(InitLayerContext &context) {
}

void EagerAttentionLayer::finalize(InitLayerContext &context) {
  // Print input information
  // std::cout << "EagerAttentionLayer Input Information:" << std::endl;
  // std::cout << "  Number of inputs: " << context.getNumInputs() << std::endl;
  
  // For now, we'll set a default output dimension same as first input
  // In a real implementation, this would be based on the attention computation
  if (context.getNumInputs() > 0) {
    std::vector<TensorDim> output_dims = {context.getInputDimensions()[0]};
    context.setOutputDimensions(output_dims);
    
    // std::cout << "EagerAttentionLayer Output Information:" << std::endl;
    // std::cout << "  Number of outputs: " << output_dims.size() << std::endl;
  }
  
  // Pre-allocate temporary tensors
  // Get input dimensions
  auto const &all_dims = context.getInputDimensions();
  auto const &query_dim = all_dims[TensorParams::query];
  auto const &key_dim = all_dims[TensorParams::key];
  auto const &value_dim = all_dims[TensorParams::value];
  
  // 1. Temporary tensor for permuted key (key.transpose("0:2:1"))
  auto key_permuted_dim = key_dim.transpose("0:2:1");
  wt_idx[0] = context.requestTensor(key_permuted_dim, "key_permuted", 
                                    Initializer::NONE, false, 
                                    TensorLifespan::ITERATION_LIFESPAN);
  
  // 2. Temporary tensor for attention scores (query.dotBatched(key_permuted))
  auto attention_scores_dim = query_dim;
  attention_scores_dim.width(key_dim.height()); // Result of matmul
  wt_idx[1] = context.requestTensor(attention_scores_dim, "attention_scores", 
                                    Initializer::NONE, false, 
                                    TensorLifespan::ITERATION_LIFESPAN);
  
  // 3. Temporary tensor for attention weights (after softmax)
  wt_idx[2] = context.requestTensor(attention_scores_dim, "attention_weights", 
                                    Initializer::NONE, false, 
                                    TensorLifespan::ITERATION_LIFESPAN);
  
  // 4. Temporary tensor for context layer (attention_weights.dotBatched(value))
  auto context_layer_dim = query_dim;
  context_layer_dim.width(value_dim.width()); // Result of matmul
  wt_idx[3] = context.requestTensor(context_layer_dim, "context_layer", 
                                    Initializer::NONE, false, 
                                    TensorLifespan::ITERATION_LIFESPAN);
  
  // 5. Temporary tensor for attention mask
  // Create mask dimension to match attention_scores: (batch, channel, query_height, key_height)
  auto attention_mask_dim = TensorDim({1, 1, query_dim.height(), key_dim.height()});
  wt_idx[4] = context.requestTensor(attention_mask_dim, "attention_mask", 
                                    Initializer::NONE, false, 
                                    TensorLifespan::ITERATION_LIFESPAN);
  
  // 6. KV Cache tensors
  wt_idx[TensorParams::cache_key] = context.requestTensor(key_dim, "cache_key",
                                    Initializer::NONE, true,
                                    TensorLifespan::MAX_LIFESPAN);
  
  wt_idx[TensorParams::cache_value] = context.requestTensor(value_dim, "cache_value",
                                    Initializer::NONE, true,
                                    TensorLifespan::MAX_LIFESPAN);
  
  wt_idx[TensorParams::present_key] = context.requestTensor(key_dim, "present_key",
                                    Initializer::NONE, true,
                                    TensorLifespan::MAX_LIFESPAN);
  
  wt_idx[TensorParams::present_value] = context.requestTensor(value_dim, "present_value",
                                    Initializer::NONE, true,
                                    TensorLifespan::MAX_LIFESPAN);
  
  // Initialize softmax activation function
  auto data_type = context.getActivationDataType();
  if (data_type == ml::train::TensorDim::DataType::FP32) {
    sm.setActiFunc<float>(ActivationType::ACT_SOFTMAX);
  } else if (data_type == ml::train::TensorDim::DataType::FP16) {
#ifdef ENABLE_FP16
    sm.setActiFunc<_FP16>(ActivationType::ACT_SOFTMAX);
#else
    throw std::runtime_error("enable-fp16 is not enabled");
#endif
  }
}

void EagerAttentionLayer::forwarding(RunLayerContext &context, bool training) {
  // std::cout << "------------------------------Attention called-------------------------------" << std::endl;
  // Get input tensors using proper indices from TensorParams
  Tensor &query = context.getInput(TensorParams::query);
  Tensor &key = context.getInput(TensorParams::key);
  Tensor &value = context.getInput(TensorParams::value);
  
  // Get output tensor
  Tensor &output = context.getOutput(0);
  
  // Get pre-allocated temporary tensors
  Tensor &key_permuted = context.getTensor(wt_idx[0]);
  Tensor &attention_scores = context.getTensor(wt_idx[1]);
  Tensor &attention_weights = context.getTensor(wt_idx[2]);
  Tensor &context_layer = context.getTensor(wt_idx[3]);
  
  // Get KV cache tensors
  Tensor &cache_key = context.getTensor(wt_idx[TensorParams::cache_key]);
  Tensor &cache_value = context.getTensor(wt_idx[TensorParams::cache_value]);
  Tensor &present_key = context.getTensor(wt_idx[TensorParams::present_key]);
  Tensor &present_value = context.getTensor(wt_idx[TensorParams::present_value]);

  // Check if we have cached data (for incremental inference)
  bool has_cached_data = (cache_key.size() > 0 && cache_value.size() > 0 && 
                         cache_key.height() > 0 && cache_value.height() > 0);
  
  if (has_cached_data) {
    // Use cached key and value
    // 1. Permute cached key with direction: 1,3,2 (0-indexed: 0,2,1)
    cache_key.transpose("0:2:1", key_permuted);
    
    // 2. MatMul query and permuted cached key
    unsigned int batch = query.batch();
    unsigned int channel = query.channel();
    unsigned int height0 = query.height();
    unsigned int width0 = query.width();
    
    unsigned int height1 = key_permuted.height();
    unsigned int width1 = key_permuted.width();
    
    query.reshape(TensorDim({batch * channel, 1, height0, width0}));
    key_permuted.reshape(TensorDim({batch * channel, 1, height1, width1}));
    attention_scores.reshape(TensorDim({batch * channel, 1, height0, width1}));
    
    query.dotBatched(key_permuted, attention_scores);
    
    attention_scores.reshape(TensorDim({batch, channel, height0, width1}));
    
    // 3. Mul (scale by sqrt of head size)
    float scale = 1.0f / sqrt(static_cast<float>(cache_key.getDim().width()));
    attention_scores.multiply_i(scale);
    
    // 4. Create and apply attention mask
    Tensor &attention_mask = context.getTensor(wt_idx[4]);
    
    // Initialize mask with large negative values
    attention_mask.setValue(-3.4028e+38f);
    
    // Apply causal mask: set diagonal and lower triangular part to 0
    // This is equivalent to torch.triu(causal_mask, diagonal=1) in PyTorch
    unsigned int mask_height = attention_scores.height();
    unsigned int mask_width = attention_scores.width();
    
    for (unsigned int i = 0; i < mask_height; ++i) {
      for (unsigned int j = 0; j <= i && j < mask_width; ++j) {
        attention_mask.setValue(0, 0, i, j, 0.0f);  // Set to 0 for non-masked positions
      }
    }
    
    // Add mask to attention scores
    attention_scores.add_i(attention_mask);
    
    // 5. Softmax activation
    sm.run_fn(attention_scores, attention_weights);
    
    // 6. MatMul softmax result with cached value
    batch = attention_weights.batch();
    channel = attention_weights.channel();
    height0 = attention_weights.height();
    width0 = attention_weights.width();
    
    height1 = cache_value.height();
    width1 = cache_value.width();
    
    attention_weights.reshape(TensorDim({batch * channel, 1, height0, width0}));
    cache_value.reshape(TensorDim({batch * channel, 1, height1, width1}));
    context_layer.reshape(TensorDim({batch * channel, 1, height0, width1}));
    
    attention_weights.dotBatched(cache_value, context_layer);
    context_layer.reshape(TensorDim({batch, channel, height0, width1}));
    
    // 7. Permute output with direction: 2,1,3 (0-indexed: 1,0,2)
    context_layer.transpose("1:0:2", output);
    
    // 8. Update present key and value
    present_key.copyData(cache_key);
    present_value.copyData(cache_value);
  } else {
    // Standard attention computation
    // 1. Permute key tensor with direction: 1,3,2 (0-indexed: 0,2,1)
    key.transpose("0:2:1", key_permuted);
    // std::cout << "------Key transpose" << std::endl;
    // std::cout << key_permuted;
    
    // 2. MatMul query and permuted key
    unsigned int batch = query.batch();
    unsigned int channel = query.channel();
    unsigned int height0 = query.height();
    unsigned int width0 = query.width();
    
    unsigned int height1 = key_permuted.height();
    unsigned int width1 = key_permuted.width();
    
    query.reshape(TensorDim({batch * channel, 1, height0, width0}));
    key_permuted.reshape(TensorDim({batch * channel, 1, height1, width1}));
    attention_scores.reshape(TensorDim({batch * channel, 1, height0, width1}));
    
    query.dotBatched(key_permuted, attention_scores);
    
    attention_scores.reshape(TensorDim({batch, channel, height0, width1}));
    // std::cout << "------Attention scores" << std::endl;
    // std::cout << attention_scores;
    
    // 3. Mul (scale by sqrt of head size)
    float scale = 1.0f / sqrt(static_cast<float>(key.getDim().width()));
    attention_scores.multiply_i(scale);
    // std::cout << "------Scaled Attention scores" << std::endl;
    // std::cout << attention_scores;
    
    // 4. Create and apply attention mask
    Tensor &attention_mask = context.getTensor(wt_idx[4]);
    
    // Initialize mask with large negative values
    attention_mask.setValue(-3.4028e+38f);
    
    // Apply causal mask: set diagonal and lower triangular part to 0
    // This is equivalent to torch.triu(causal_mask, diagonal=1) in PyTorch
    unsigned int mask_height = attention_scores.height();
    unsigned int mask_width = attention_scores.width();
    
    for (unsigned int i = 0; i < mask_height; ++i) {
      for (unsigned int j = 0; j <= i && j < mask_width; ++j) {
        attention_mask.setValue(0, 0, i, j, 0.0f);  // Set to 0 for non-masked positions
      }
    }
    
    // std::cout << "------Attention Mask" << std::endl;
    // std::cout << attention_mask;
    
    // Add mask to attention scores
    attention_scores.add_i(attention_mask);
    
    // std::cout << "------Masked Scaled Attention scores" << std::endl;
    // std::cout << attention_scores;
    
    // 5. Softmax activation
    sm.run_fn(attention_scores, attention_weights);
    // std::cout << "------Attention weights" << std::endl;
    // std::cout << attention_weights;
    
    // 6. MatMul softmax result with value
    batch = attention_weights.batch();
    channel = attention_weights.channel();
    height0 = attention_weights.height();
    width0 = attention_weights.width();
    
    height1 = value.height();
    width1 = value.width();
    
    attention_weights.reshape(TensorDim({batch * channel, 1, height0, width0}));
    value.reshape(TensorDim({batch * channel, 1, height1, width1}));
    context_layer.reshape(TensorDim({batch * channel, 1, height0, width1}));
    
    attention_weights.dotBatched(value, context_layer);
    context_layer.reshape(TensorDim({batch, channel, height0, width1}));
    // std::cout << "------Context layer" << std::endl;
    // std::cout << context_layer;
    
    // 7. Permute output with direction: 2,1,3 (0-indexed: 1,0,2)
    context_layer.transpose("1:0:2", output);
    // std::cout << "------Attention Output" << std::endl;
    // std::cout << output;
    
    // 8. Update present key and value with current key and value
    present_key.copyData(key);
    present_value.copyData(value);
  }
  // std::cout << "--------------------------------------------------------------------------------" << std::endl;
}

void EagerAttentionLayer::incremental_forwarding(RunLayerContext &context,
                                            unsigned int from, unsigned int to,
                                            bool training) {
  // Get input tensors using proper indices from TensorParams
  Tensor &query = context.getInput(TensorParams::query);
  Tensor &key = context.getInput(TensorParams::key);
  Tensor &value = context.getInput(TensorParams::value);
  
  // Get output tensor
  Tensor &output = context.getOutput(0);
  
  // Get pre-allocated temporary tensors
  Tensor &key_permuted = context.getTensor(wt_idx[0]);
  Tensor &attention_scores = context.getTensor(wt_idx[1]);
  Tensor &attention_weights = context.getTensor(wt_idx[2]);
  Tensor &context_layer = context.getTensor(wt_idx[3]);
  
  // Get KV cache tensors
  Tensor &cache_key = context.getTensor(wt_idx[TensorParams::cache_key]);
  Tensor &cache_value = context.getTensor(wt_idx[TensorParams::cache_value]);
  Tensor &present_key = context.getTensor(wt_idx[TensorParams::present_key]);
  Tensor &present_value = context.getTensor(wt_idx[TensorParams::present_value]);

  // Get dimensions
  TensorDim query_dim = query.getDim();
  TensorDim key_dim = key.getDim();
  TensorDim value_dim = value.getDim();
  
  // Create step dimensions
  TensorDim query_step_dim = {query_dim.batch(), query_dim.channel(), to - from, query_dim.width()};
  TensorDim key_step_dim = {key_dim.batch(), key_dim.channel(), to, key_dim.width()};
  TensorDim value_step_dim = {value_dim.batch(), value_dim.channel(), to, value_dim.width()};
  
  // Create step tensors
  Tensor query_step = query.getSharedDataTensor(query_step_dim, from * query_dim.width(), true);
  Tensor key_step = key.getSharedDataTensor(key_step_dim, 0, true);
  Tensor value_step = value.getSharedDataTensor(value_step_dim, 0, true);
  
  // Create cache step dimensions
  TensorDim cache_key_step_dim = {key_dim.batch(), key_dim.channel(), to - from, key_dim.width()};
  TensorDim cache_value_step_dim = {value_dim.batch(), value_dim.channel(), to - from, value_dim.width()};
  
  // Create cache step tensors
  Tensor cache_key_step = cache_key.getSharedDataTensor(cache_key_step_dim, from * key_dim.width(), true);
  Tensor cache_value_step = cache_value.getSharedDataTensor(cache_value_step_dim, from * value_dim.width(), true);
  
  // Create cached dimensions
  TensorDim cached_key_dim = {key_dim.batch(), key_dim.channel(), to, key_dim.width(), key.getTensorType()};
  TensorDim cached_value_dim = {value_dim.batch(), value_dim.channel(), to, value_dim.width(), value.getTensorType()};
  Tensor cached_key = cache_key.getSharedDataTensor(cached_key_dim, 0, true);
  Tensor cached_value = cache_value.getSharedDataTensor(cached_value_dim, 0, true);
  
  // Create output step dimensions
  TensorDim output_dim = output.getDim();
  TensorDim output_step_dim = {output_dim.batch(), output_dim.channel(), to - from, output_dim.width()};
  Tensor output_step = output.getSharedDataTensor(output_step_dim, from * output_dim.width(), true);
  
  // Create attention scores step dimensions
  TensorDim attention_scores_dim = attention_scores.getDim();
  TensorDim attention_scores_step_dim = {attention_scores_dim.batch(), attention_scores_dim.channel(), to - from, to};
  Tensor attention_scores_step = attention_scores.getSharedDataTensor(attention_scores_step_dim, from * attention_scores_dim.width(), true);
  
  // Create attention weights step dimensions
  TensorDim attention_weights_step_dim = {attention_scores_step_dim.batch(), attention_scores_step_dim.channel(), attention_scores_step_dim.height(), attention_scores_step_dim.width()};
  Tensor attention_weights_step = attention_weights.getSharedDataTensor(attention_weights_step_dim, from * attention_scores_dim.width(), true);
  
  // Create context layer step dimensions
  TensorDim context_layer_dim = context_layer.getDim();
  TensorDim context_layer_step_dim = {context_layer_dim.batch(), context_layer_dim.channel(), to - from, context_layer_dim.width()};
  Tensor context_layer_step = context_layer.getSharedDataTensor(context_layer_step_dim, from * context_layer_dim.width(), true);
  
  // 1. Copy key and value to cache
  cache_key_step.copyData(key_step);
  cache_value_step.copyData(value_step);
  
  // 2. Permute cached key with direction: 1,3,2 (0-indexed: 0,2,1)
  cached_key.transpose("0:2:1", key_permuted);
  
  // 3. MatMul query and permuted cached key
  unsigned int batch = query_step.batch();
  unsigned int channel = query_step.channel();
  unsigned int height0 = query_step.height();
  unsigned int width0 = query_step.width();
  
  unsigned int height1 = key_permuted.height();
  unsigned int width1 = key_permuted.width();
  
  query_step.reshape(TensorDim({batch * channel, 1, height0, width0}));
  key_permuted.reshape(TensorDim({batch * channel, 1, height1, width1}));
  attention_scores_step.reshape(TensorDim({batch * channel, 1, height0, width1}));
  
  query_step.dotBatched(key_permuted, attention_scores_step);
  
  attention_scores_step.reshape(TensorDim({batch, channel, height0, width1}));
  
  // 4. Mul (scale by sqrt of head size)
  float scale = 1.0f / sqrt(static_cast<float>(cached_key.getDim().width()));
  attention_scores_step.multiply_i(scale);
  
  // 5. Create and apply attention mask
  Tensor &attention_mask = context.getTensor(wt_idx[4]);
  
  // Initialize mask with large negative values
  attention_mask.setValue(-3.4028e+38f);
  
  // Apply causal mask: set diagonal and lower triangular part to 0
  unsigned int mask_height = attention_scores_step.height();
  unsigned int mask_width = attention_scores_step.width();
  
  for (unsigned int i = 0; i < mask_height; ++i) {
    for (unsigned int j = 0; j <= i && j < mask_width; ++j) {
      attention_mask.setValue(0, 0, i, j, 0.0f);  // Set to 0 for non-masked positions
    }
  }
  
  // Add mask to attention scores
  attention_scores_step.add_i(attention_mask);
  
  // 6. Softmax activation
  sm.run_fn(attention_scores_step, attention_weights_step);
  
  // 7. MatMul softmax result with cached value
  batch = attention_weights_step.batch();
  channel = attention_weights_step.channel();
  height0 = attention_weights_step.height();
  width0 = attention_weights_step.width();
  
  height1 = cached_value.height();
  width1 = cached_value.width();
  
  attention_weights_step.reshape(TensorDim({batch * channel, 1, height0, width0}));
  cached_value.reshape(TensorDim({batch * channel, 1, height1, width1}));
  context_layer_step.reshape(TensorDim({batch * channel, 1, height0, width1}));
  
  attention_weights_step.dotBatched(cached_value, context_layer_step);
  context_layer_step.reshape(TensorDim({batch, channel, height0, width1}));
  
  // 8. Permute output with direction: 2,1,3 (0-indexed: 1,0,2)
  context_layer_step.transpose("1:0:2", output_step);
  
  // 9. Update present key and value
  present_key.copyData(cached_key);
  present_value.copyData(cached_value);
}
